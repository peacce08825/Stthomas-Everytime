rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Each user's profile + timetables.
    // Any signed-in user can read (needed to view friends' schedules);
    // only you can write your own.
    match /users/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // Anonymous board.
    // Signed-in users can read, post, and add comments;
    // only the original poster can delete their post.
    match /board/{postId} {
      allow read, create: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if request.auth != null && request.auth.uid == resource.data.uid;
    }
  }
}
